# КОМАНДЫ ДЛЯ PYTHONANYWHERE (Bash Console)

# 1. Перейти в директорию проекта
cd /home/hyperstls/trudnik

# 2. Привязать к репозиторию (если не привязан)
git remote add origin https://github.com/Hyperstls/trudnik.git

# 3. Скачать изменения
git pull origin main

# 4. Создать резервную копию (на всякий случай)
cp app.py app.py.backup.20260603

# 5. Перезапустить приложение
touch app.py.wsgi

# 6. Проверить статус сервера
curl -s https://hyperstls.pythonanywhere.com | head -20

# 7. Проверить логи (если ошибка)
cat /var/log/hyperstls.pythonanywhere.com.error.log
